FROM python:3.12-slim-bookworm

RUN apt-get update \
    && apt-get install -y --no-install-recommends ffmpeg unzip ca-certificates \
    && rm -rf /var/lib/apt/lists/*

WORKDIR /app

# O repo no GitHub tem só este zip (enviado do iPhone).
COPY amethora.zip /tmp/amethora.zip
RUN unzip -o /tmp/amethora.zip -d /tmp/src \
    && if [ -d /tmp/src/amethora ]; then cp -a /tmp/src/amethora/. /app/; \
       else cp -a /tmp/src/. /app/; fi \
    && rm -rf /tmp/src /tmp/amethora.zip \
    && pip install --no-cache-dir -r /app/requirements.txt

ENV PYTHONUNBUFFERED=1
EXPOSE 8080

CMD ["sh", "-c", "uvicorn webapp.app:app --host 0.0.0.0 --port ${PORT:-8080}"]
